{% comment %} Renders a single tag as a linked badge. If the tag has an entry in _data/project_tags.yml, links to the project page. Otherwise links to /tag/tagname. Usage: {% include tag_link.html tag=tag_name %} {% endcomment %} {% assign project_url = site.data.project_tags[include.tag] %} {% if project_url %} {{ include.tag }}  {% else %} {{ include.tag }}  {% endif %}
